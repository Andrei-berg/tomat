    Create src/lib/session.ts:
    ```typescript
    import 'server-only'
    import { getIronSession, SessionOptions } from 'iron-session'
    import { cookies } from 'next/headers'

    export interface SessionData {
      isAuthenticated: boolean
    }

    export const sessionOptions: SessionOptions = {
      password: process.env.IRON_SESSION_SECRET!,
      cookieName: '_tomat_session',
      cookieOptions: {
        secure: process.env.NODE_ENV === 'production',
        sameSite: 'lax',
        httpOnly: true,
      },
    }

    export async function getSession() {
      const cookieStore = await cookies()
      return getIronSession<SessionData>(cookieStore, sessionOptions)
    }

    export async function createSession() {
      const session = await getSession()
      session.isAuthenticated = true
      await session.save()
    }

    export async function deleteSession() {
      const session = await getSession()
      session.destroy()
    }
    ```

    Create src/lib/dal.ts:
    ```typescript
    import 'server-only'
    import { cache } from 'react'
    import { redirect } from 'next/navigation'
    import { getSession } from '@/lib/session'

    export const verifySession = cache(async () => {
      const session = await getSession()
      if (!session.isAuthenticated) {
        redirect('/login')
      }
      return { isAuth: true }
    })
    ```

    CRITICAL NOTES:
    - Both files MUST start with `import 'server-only'` — prevents session code from leaking to client bundle
    - `cookies()` MUST be awaited (Next.js 15 breaking change — cookies() returns Promise)
    - `IRON_SESSION_SECRET` must be >= 32 chars — iron-session throws at runtime if shorter
    - Use `session.destroy()` for deleteSession (not `session.isAuthenticated = false`)
