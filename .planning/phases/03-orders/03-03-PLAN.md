**Логика:**
1. `await verifySession()`
2. Распаковать searchParams: `const { date: dateParam } = await searchParams`
3. Вычислить целевую дату:
   ```typescript
   const today = new Date().toISOString().split('T')[0]
   const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0]
   const targetDate = dateParam ?? today
   ```
4. `const orders = await getOrdersByDate(targetDate)`
5. Вычислить итог дня:
   ```typescript
   function getEffective(o: OrderRow): number {
     if (o.manual_total != null) return o.manual_total
     const base = o.calculated_total ?? 0
     if (o.discount_percent) return Math.round(base * (1 - o.discount_percent / 100) * 100) / 100
     return base
   }
   const dayTotals = { cash: 0, card: 0, debt: 0 }
   for (const o of orders) { dayTotals[o.payment_type] += getEffective(o) }
   ```
